### 1. Adaptador UI y Gráficos (`public/js/ui-semana2.js`)

Este módulo toma el resultado de `CasosEstudio` y renderiza automáticamente:

1. Los **dos gráficos** exigidos por módulo (Trayectoria/Curva y Error/Convergencia) usando Chart.js.
2. Las **tablas de iteraciones** con criterio de parada.
3. Las **dos conclusiones separadas** (Matemática y del Problema).
4. Los botones de **Guardar en SQLite** y **Relanzar / Verificar Coincidencia** desde el Historial.

```javascript
import { CasosEstudio } from "./casos.js";
import { PersistenciaAPI } from "./persistencia.js";

let chartPrincipal = null;
let chartErrores = null;
let simulacionActual = null;

export function inicializarPanelSemana2() {
  const btnEjecutar = document.getElementById("btn-ejecutar-s2");
  const btnGuardar = document.getElementById("btn-guardar-s2");
  const btnHistorial = document.getElementById("btn-cargar-historial");

  if (btnEjecutar) {
    btnEjecutar.addEventListener("click", () => {
      const casoId = Number(document.getElementById("select-caso").value);
      const tipo = document.getElementById("select-escenario").value;
      ejecutarYRenderizar(casoId, tipo);
    });
  }

  if (btnGuardar) {
    btnGuardar.addEventListener("click", async () => {
      if (!simulacionActual) return alert("Primero ejecuta una simulación.");
      const resp = await PersistenciaAPI.guardarSimulacion(simulacionActual);
      alert(`Simulación guardada en SQLite con ID #${resp.simulacion_id}`);
      await refrescarHistorial();
    });
  }

  if (btnHistorial) {
    btnHistorial.addEventListener("click", refrescarHistorial);
  }
}

export function ejecutarYRenderizar(casoId, tipo) {
  const mapaCasos = {
    1: () => CasosEstudio.detencionEmergencia(tipo),
    2: () => CasosEstudio.deterioroMecanico(tipo),
    3: () => CasosEstudio.oscilacionesViento(tipo),
    4: () => CasosEstudio.reconstruccionSensores(tipo)
  };

  simulacionActual = mapaCasos[casoId]();
  renderizarTablas(simulacionActual);
  renderizarConclusiones(simulacionActual);
  renderizarGraficos(simulacionActual);
  return simulacionActual;
}

function renderizarTablas(sim) {
  const contenedor = document.getElementById("contenedor-tablas-s2");
  if (!contenedor) return;

  contenedor.innerHTML = sim.ejecuciones.map((ej, idx) => {
    const filas = ej.iteraciones.slice(0, 15).map(it => `
      <tr>
        <td>${it.numero}</td>
        <td><code>${JSON.stringify(it.aproximacion)}</code></td>
        <td>${it.residual !== null ? it.residual.toExponential(3) : "—"}</td>
        <td>${it.error_abs !== null ? it.error_abs.toExponential(3) : "—"}</td>
        <td>${it.error_rel !== null ? (it.error_rel * 100).toFixed(4) + "%" : "N/A"}</td>
      </tr>
    `).join("");

    return `
      <details ${idx === 0 ? "open" : ""} class="bloque-metodo">
        <summary>
          <strong>${ej.metodo}</strong> —
          Estado: <span class="${ej.convergio ? "ok" : "warn"}">${ej.convergio ? "Convergió" : "No convergió"}</span> |
          Tiempo: ${ej.tiempo_ms.toFixed(2)} ms
        </summary>
        <p><small>Referencia: ${ej.resumen.referencia}</small></p>
        <table>
          <thead>
            <tr><th>Iter</th><th>Aproximación</th><th>Residual</th><th>Error Abs.</th><th>Error Rel.</th></tr>
          </thead>
          <tbody>${filas}</tbody>
        </table>
      </details>
    `;
  }).join("");
}

function renderizarConclusiones(sim) {
  const interp = sim.ejecuciones[0]?.resumen?.interpretacion;
  if (!interp) return;

  const elMat = document.getElementById("conclusion-matematica");
  const elProb = document.getElementById("conclusion-problema");
  if (elMat) elMat.textContent = interp.matematica;
  if (elProb) elProb.textContent = interp.problema;
}

function renderizarGraficos(sim) {
  const ctx1 = document.getElementById("grafico-escenario")?.getContext("2d");
  const ctx2 = document.getElementById("grafico-errores")?.getContext("2d");
  if (!ctx1 || !ctx2) return;

  if (chartPrincipal) chartPrincipal.destroy();
  if (chartErrores) chartErrores.destroy();

  // Gráfico 1: Solución / Escenario
  let labels = [];
  let datasets1 = [];

  if (sim.caso_id === 1) {
    labels = sim.graficos.serie_principal.map(p => p.t.toFixed(2));
    datasets1 = [
      { label: "Posición x(t) RK4 [m]", data: sim.graficos.serie_principal.map(p => p.x_rk4), borderColor: "#2563eb", fill: false },
      { label: "Velocidad v(t) RK4 [m/s]", data: sim.graficos.serie_principal.map(p => p.v_rk4), borderColor: "#dc2626", fill: false }
    ];
  } else if (sim.caso_id === 2) {
    labels = sim.graficos.curva_deterioro.map(p => p.t);
    datasets1 = [
      { label: "Deterioro D(t)", data: sim.graficos.curva_deterioro.map(p => p.D), borderColor: "#d97706", fill: false },
      { label: "Umbral Crítico", data: sim.graficos.curva_deterioro.map(p => p.umbral), borderColor: "#dc2626", borderDash: [5, 5], fill: false }
    ];
  } else if (sim.caso_id === 3) {
    labels = sim.graficos.oscilacion_rk4.map(p => p.t.toFixed(2));
    datasets1 = [
      { label: "Desplazamiento RK4 [m]", data: sim.graficos.oscilacion_rk4.map(p => p.x_rk4), borderColor: "#059669", fill: false },
      { label: "Desplazamiento RK2 [m]", data: sim.graficos.oscilacion_rk2.map(p => p.x_rk2), borderColor: "#9333ea", borderDash: [4, 4], fill: false }
    ];
  } else {
    labels = sim.graficos.reconstruccion.map(p => p.t.toFixed(2));
    datasets1 = [
      { label: "Señal Real", data: sim.graficos.reconstruccion.map(p => p.real), borderColor: "#111827", fill: false },
      { label: "Lagrange", data: sim.graficos.reconstruccion.map(p => p.lagrange), borderColor: "#dc2626", borderDash: [4, 4], fill: false },
      { label: "Splines Cúbicos", data: sim.graficos.reconstruccion.map(p => p.spline), borderColor: "#2563eb", fill: false }
    ];
  }

  chartPrincipal = new Chart(ctx1, {
    type: "line",
    data: { labels, datasets: datasets1 },
    options: { responsive: true, plugins: { title: { display: true, text: `Escenario ${sim.tipo_escenario} — Dinámica del Modelo` } } }
  });

  // Gráfico 2: Comparación de Error / Convergencia por Método
  const metodosConError = sim.ejecuciones.filter(e => e.iteraciones.some(it => it.error_abs !== null));
  const maxLen = Math.max(...metodosConError.map(m => m.iteraciones.length), 1);
  const iterLabels = Array.from({ length: maxLen }, (_, i) => `It ${i + 1}`);
  const colores = ["#2563eb", "#dc2626", "#059669", "#d97706", "#7c3aed"];

  chartErrores = new Chart(ctx2, {
    type: "line",
    data: {
      labels: iterLabels,
      datasets: metodosConError.map((m, idx) => ({
        label: `${m.metodo} (Error Abs.)`,
        data: m.iteraciones.map(it => it.error_abs),
        borderColor: colores[idx % colores.length],
        fill: false
      }))
    },
    options: {
      responsive: true,
      scales: { y: { type: "logarithmic", title: { display: true, text: "Error Absoluto (escala log)" } } },
      plugins: { title: { display: true, text: "Convergencia y Comparación de Errores" } }
    }
  });
}

export async function refrescarHistorial() {
  const lista = await PersistenciaAPI.listarHistorial();
  const tbody = document.getElementById("tbody-historial");
  if (!tbody) return;

  tbody.innerHTML = lista.map(row => `
    <tr>
      <td>#${row.id}</td>
      <td>${row.caso_nombre}</td>
      <td>${row.escenario_tipo}</td>
      <td>${row.fecha}</td>
      <td>
        <button data-relanzar="${row.id}">Recuperar y Verificar</button>
      </td>
    </tr>
  `).join("");

  tbody.querySelectorAll("button[data-relanzar]").forEach(btn => {
    btn.addEventListener("click", async () => {
      const id = Number(btn.dataset.relanzar);
      const guardada = await PersistenciaAPI.obtenerSimulacion(id);
      const recalculada = ejecutarYRenderizar(guardada.caso_id, guardada.escenario_tipo);

      const v1 = guardada.ejecuciones[0]?.resumen?.valor;
      const v2 = recalculada.ejecuciones[0]?.resumen?.valor;
      const valNum1 = Array.isArray(v1) ? v1[0] : (typeof v1 === "object" ? v1.error_max : v1);
      const valNum2 = Array.isArray(v2) ? v2[0] : (typeof v2 === "object" ? v2.error_max : v2);

      const check = PersistenciaAPI.verificarCoincidencia(valNum1, valNum2, 1e-9);
      alert(
        `Prueba de Reproducibilidad (ID #${id}):\n` +
        `Guardado: ${valNum1}\nRecalculado: ${valNum2}\n` +
        `Diferencia: ${check.diferencia.toExponential(2)} | Coincide (tol 1e-9): ${check.coincide ? "SÍ" : "NO"}`
      );
    });
  });
}
```

### 3. Script de Verificación Analítica y Persistencia (`tests/verificar_semana2.js`)

Ejecuta `bun test tests/verificar_semana2.js` para generar la evidencia numérica de las 4 familias contra soluciones analíticas cerradas y validar el ciclo completo de guardado/recuperación en SQLite para tu informe de experimentos:

```javascript
import { describe, expect, test } from "bun:test";
import { Raices } from "../public/js/numerics/raices.js";
import { Interpolacion } from "../public/js/numerics/interpolacion.js";
import { Integracion } from "../public/js/numerics/integracion.js";
import { EDO } from "../public/js/numerics/edo.js";
import { CasosEstudio } from "../public/js/casos.js";

describe("Verificación Analítica — SIMTELE-RISK Semana 2", () => {
  test("1. Raíces: cos(x) - x = 0 (Dottie number ≈ 0.7390851332151607)", () => {
    const ref = 0.7390851332151607;
    const f = (x) => Math.cos(x) - x;
    const df = (x) => -Math.sin(x) - 1;

    const bis = Raices.biseccion(f, 0, 1, { tol: 1e-8, refAnalitica: ref });
    const newt = Raices.newtonRaphson(f, df, 0.5, { tol: 1e-10, refAnalitica: ref });
    const sec = Raices.secante(f, 0, 1, { tol: 1e-10, refAnalitica: ref });

    expect(bis.metrica_error.error_abs).toBeLessThan(1e-7);
    expect(newt.metrica_error.error_abs).toBeLessThan(1e-10);
    expect(sec.metrica_error.error_abs).toBeLessThan(1e-10);
  });

  test("2. Integración: ∫_0^π sin(x) dx = 2", () => {
    const f = Math.sin;
    const trap = Integracion.trapecioCompuesto(f, 0, Math.PI, 1000, { refAnalitica: 2 });
    const simp = Integracion.simpson13Compuesto(f, 0, Math.PI, 100, { refAnalitica: 2 });
    const romb = Integracion.romberg(f, 0, Math.PI, { tol: 1e-10, refAnalitica: 2 });

    expect(trap.metrica_error.error_abs).toBeLessThan(1e-5);
    expect(simp.metrica_error.error_abs).toBeLessThan(1e-8);
    expect(romb.metrica_error.error_abs).toBeLessThan(1e-10);
  });

  test("3. Interpolación: reconstrucción de e^(-t) en puntos intermedios", () => {
    const xs = [0, 0.5, 1.0, 1.5, 2.0];
    const ys = xs.map(t => Math.exp(-t));
    const evalPts = [0.25, 0.75, 1.25];
    const refs = evalPts.map(t => Math.exp(-t));

    const lag = Interpolacion.lagrange(xs, ys, evalPts, { refsAnaliticas: refs });
    const newt = Interpolacion.newton(xs, ys, evalPts, { refsAnaliticas: refs });
    const spl = Interpolacion.splinesCubicos(xs, ys, evalPts, { refsAnaliticas: refs });

    expect(Math.abs(lag.predicciones[0] - refs[0])).toBeLessThan(1e-3);
    expect(Math.abs(newt.predicciones[0] - lag.predicciones[0])).toBeLessThan(1e-12);
    expect(Math.abs(spl.predicciones[1] - refs[1])).toBeLessThan(1e-2);
  });

  test("4. EDO: y' = -2y, y(0) = 1 -> y(1) = e^(-2)", () => {
    const F = (t, [y]) => [-2 * y];
    const refFn = (t) => [Math.exp(-2 * t)];

    const rk2 = EDO.rk2Heun(F, 0, [1], 1, 0.05, { refAnaliticaFn: refFn });
    const rk4 = EDO.rk4Clasico(F, 0, [1], 1, 0.05, { refAnaliticaFn: refFn });

    expect(rk2.metrica_error.error_abs).toBeLessThan(1e-3);
    expect(rk4.metrica_error.error_abs).toBeLessThan(1e-7);
  });

  test("5. Cobertura matriz 4 módulos × 3 escenarios (12 ejecuciones válidas)", () => {
    const escenarios = ["NORMAL", "ESTRES", "MAL_CONDICIONADO"];
    for (const esc of escenarios) {
      expect(CasosEstudio.detencionEmergencia(esc).ejecuciones.length).toBeGreaterThan(0);
      expect(CasosEstudio.deterioroMecanico(esc).ejecuciones.length).toBeGreaterThan(0);
      expect(CasosEstudio.oscilacionesViento(esc).ejecuciones.length).toBeGreaterThan(0);
      expect(CasosEstudio.reconstruccionSensores(esc).ejecuciones.length).toBeGreaterThan(0);
    }
  });
});
```
